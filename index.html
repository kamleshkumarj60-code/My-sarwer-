const express = require("express"); const path = require("path"); const app = express(); const PORT = process.env.PORT || 3000; app.use(express.json()); app.use(express.static(path.join(__dirname, "public"))); let period = 100001; let history = []; function makeResult() { const number = Math.floor(Math.random() * 10); let size = number >= 5 ? "BIG" : "SMALL"; let color; if (number === 0 || number === 5) color = "VIOLET"; else if (number % 2 === 0) color = "RED"; else color = "GREEN"; return { number, size, color }; } app.get("/api/result", (req, res) => { const result = makeResult(); const data = { period: String(period++), time: new Date().toLocaleTimeString("en-IN"), ...result }; history.unshift(data); history = history.slice(0, 20); res.json(data); }); app.get("/api/history", (req, res) => { res.json(history); }); app.get("*", (req, res) => { res.sendFile(path.join(__dirname, "public", "index.html")); }); app.listen(PORT, () => { console.log(`Demo server running on port ${PORT}`); });
